    return match (s) {

    };

func act(s: Shape) -> void {
    match (s) {
        Shape::Nothing => { io.out("none"); }
        _ => { io.out("some"); }

func main() -> i8 {
    let x: i64 = -2 ** 2;
    for (let i: i32 = 0; i < 10; i++) {
        if (i % 2 == 0) { continue; }
        elseif (i > 7) { break; }
        else { x += i as i64; }
    }
    do { x--; } while (x > 0);
    switch (x) {
        case 1: break;
        case 2:
        case 3: x = 0; break;
        default: x = 1;
    }
    return x as i8;
}
